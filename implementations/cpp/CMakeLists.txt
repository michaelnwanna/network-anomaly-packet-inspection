cmake_minimum_required(VERSION 3.24)
project(traceglass_cpp VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL 3.0...<4 REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
include(FetchContent)
find_package(Boost 1.74 QUIET)
if(NOT Boost_FOUND)
  FetchContent_Declare(boost URL https://archives.boost.io/release/1.86.0/source/boost_1_86_0.tar.bz2)
  FetchContent_GetProperties(boost)
  if(NOT boost_POPULATED)
    FetchContent_Populate(boost)
  endif()
  set(Boost_INCLUDE_DIRS "${boost_SOURCE_DIR}")
endif()
find_package(nlohmann_json 3.11 QUIET)
if(NOT nlohmann_json_FOUND)
  # Only the single header is needed; pin its version in the bootstrap URL.
  set(JSON_HEADER "${CMAKE_BINARY_DIR}/deps/nlohmann/json.hpp")
  if(NOT EXISTS "${JSON_HEADER}")
    file(DOWNLOAD https://raw.githubusercontent.com/nlohmann/json/v3.11.3/single_include/nlohmann/json.hpp
      "${JSON_HEADER}" STATUS JSON_DOWNLOAD TLS_VERIFY ON)
    list(GET JSON_DOWNLOAD 0 JSON_RESULT)
    if(NOT JSON_RESULT EQUAL 0)
      message(FATAL_ERROR "JSON header download failed: ${JSON_DOWNLOAD}")
    endif()
  endif()
  add_library(nlohmann_json INTERFACE)
  target_include_directories(nlohmann_json INTERFACE "${CMAKE_BINARY_DIR}/deps")
  add_library(nlohmann_json::nlohmann_json ALIAS nlohmann_json)
endif()
set(CONTRACT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(EMBEDDED "${CMAKE_BINARY_DIR}/generated/embedded.hpp")
add_custom_command(OUTPUT "${EMBEDDED}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/embed.py" "${CONTRACT_ROOT}" "${EMBEDDED}"
  DEPENDS embed.py "${CONTRACT_ROOT}/internal/classify/defaults.json"
    "${CONTRACT_ROOT}/testdata/conformance/classification.json"
    "${CONTRACT_ROOT}/internal/dashboard/web/index.html"
    "${CONTRACT_ROOT}/internal/dashboard/web/app.js"
    "${CONTRACT_ROOT}/internal/dashboard/web/style.css")
add_executable(traceglass src/main.cpp "${EMBEDDED}")
target_include_directories(traceglass PRIVATE "${CMAKE_BINARY_DIR}/generated" ${Boost_INCLUDE_DIRS})
target_compile_definitions(traceglass PRIVATE BOOST_ERROR_CODE_HEADER_ONLY BOOST_SYSTEM_NO_DEPRECATED)
target_link_libraries(traceglass PRIVATE OpenSSL::SSL OpenSSL::Crypto SQLite::SQLite3 Threads::Threads nlohmann_json::nlohmann_json)
target_compile_options(traceglass PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS traceglass RUNTIME DESTINATION bin)
enable_testing()
add_test(NAME native-unit COMMAND traceglass self-test)
add_test(NAME local-integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:traceglass>")
set_tests_properties(local-integration PROPERTIES TIMEOUT 90)
